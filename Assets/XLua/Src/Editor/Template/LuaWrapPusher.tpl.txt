#else
using LuaAPI = XLua.LuaDLL.Bridge;

using LuaCSFunction = XLua.LuaDLL.lua_CSFunction;

using System;
<%
require "TemplateCommon"
%>
[assembly: XLua.Binding.GeneratedBindings(typeof(XLua.CSObjectWrap.WrapPusher), XLua.Binding.BindingApi.Version)]

namespace XLua.CSObjectWrap
{
    using Utils = XLua.Utils;

    // Push, get and update of GCOptimize types and enums without boxing, as
    // extension methods of ObjectTranslator. Each type has a slot
    // (XLua.Binding.BindingApi.AllocateSlot) for its per-LuaEnv type id.
    public static class WrapPusher
    {
        <%ForEachCsList(purevaluetypes, function(type_info)
        local slot_name = 's_' .. CSVariableName(type_info.Type) .. '_Slot'
        local full_type_name = CsFullTypeName(type_info.Type)
		local is_enum = type_info.Type.IsEnum
		local pack_class = is_enum and "CopyByValue" or "PackUnpack"
        %>static readonly int <%=slot_name%> = XLua.Binding.BindingApi.AllocateSlot(typeof(<%=full_type_name%>));

        public static void Push<%=CSVariableName(type_info.Type)%>(this ObjectTranslator translator, RealStatePtr L, <%=full_type_name%> val)
        {
            int type_id = XLua.Binding.BindingApi.TypeId(translator, L, <%=slot_name%>);
			<%if is_enum then%>
			int enum_ref = XLua.Binding.BindingApi.EnumCacheRef(translator, L, <%=slot_name%>);
			if (LuaAPI.xlua_tryget_cachedud(L, (int)val, enum_ref) == 1)
            {
			    return;
			}
			<%
			end
			if type_info.Flag == CS.XLua.OptimizeFlag.PackAsTable then
			%>
			LuaAPI.xlua_pushcstable(L, <%=type_info.FieldInfos.Count%>, type_id);
			<%ForEachCsList(type_info.FieldInfos, function(fieldInfo)%>
			LuaAPI.xlua_pushasciistring(L, "<%=fieldInfo.Name%>");
			<%=GetPushStatement(fieldInfo.Type, "val."..fieldInfo.Name)%>;
			LuaAPI.lua_rawset(L, -3);
			<%end)%>
			<%else%>
            IntPtr buff = LuaAPI.xlua_pushstruct(L, <%=is_enum and 4 or type_info.Size%>, type_id);
            if (!<%=pack_class%>.Pack(buff, 0, <%=is_enum and "(int)" or ""%>val))
            {
                throw new Exception("pack fail fail for <%=full_type_name%> ,value="+val);
            }
			<%
			end
			if is_enum then
			%>
			LuaAPI.lua_getref(L, enum_ref);
			LuaAPI.lua_pushvalue(L, -2);
			LuaAPI.xlua_rawseti(L, -2, (int)val);
			LuaAPI.lua_pop(L, 1);
			<%end%>
        }

        public static void Get<%=CSVariableName(type_info.Type)%>(this ObjectTranslator translator, RealStatePtr L, int index, out <%=full_type_name%> val)
        {
		    LuaTypes type = LuaAPI.lua_type(L, index);
            if (type == LuaTypes.LUA_TUSERDATA )
            {
			    if (LuaAPI.xlua_gettypeid(L, index) != XLua.Binding.BindingApi.TypeId(translator, L, <%=slot_name%>))
				{
				    throw new Exception("invalid userdata for <%=full_type_name%>");
				}

                IntPtr buff = LuaAPI.lua_touserdata(L, index);<%if is_enum then%>
				int e;
                <%end%>if (!<%=pack_class%>.UnPack(buff, 0, out <%=is_enum and "e" or "val"%>))
                {
                    throw new Exception("unpack fail for <%=full_type_name%>");
                }<%if is_enum then%>
				val = (<%=full_type_name%>)e;
                <%end%>
            }<%if not is_enum then%>
			else if (type ==LuaTypes.LUA_TTABLE)
			{
			    PackUnpack.UnPack(translator, L, index, out val);
			}<%end%>
            else
            {
                val = (<%=full_type_name%>)XLua.Binding.BindingApi.GetCaster(translator, typeof(<%=full_type_name%>))(L, index, null);
            }
        }

        public static void Update<%=CSVariableName(type_info.Type)%>(this ObjectTranslator translator, RealStatePtr L, int index, <%=full_type_name%> val)
        {
		    <%if type_info.Flag == CS.XLua.OptimizeFlag.PackAsTable then%>
			if (LuaAPI.lua_type(L, index) == LuaTypes.LUA_TTABLE)
            {
			    return;
			}
			<%else%>
            if (LuaAPI.lua_type(L, index) == LuaTypes.LUA_TUSERDATA)
            {
			    if (LuaAPI.xlua_gettypeid(L, index) != XLua.Binding.BindingApi.TypeId(translator, L, <%=slot_name%>))
				{
				    throw new Exception("invalid userdata for <%=full_type_name%>");
				}

                IntPtr buff = LuaAPI.lua_touserdata(L, index);
                if (!<%=pack_class%>.Pack(buff, 0,  <%=is_enum and "(int)" or ""%>val))
                {
                    throw new Exception("pack fail for <%=full_type_name%> ,value="+val);
                }
            }
			<%end%>
            else
            {
                throw new Exception("try to update a data with lua type:" + LuaAPI.lua_type(L, index));
            }
        }

        <%end)%>
		// table cast optimze
		<%ForEachCsList(tableoptimzetypes, function(type_info)
		local full_type_name = CsFullTypeName(type_info.Type)
		%>
		public static void Get<%=CSVariableName(type_info.Type)%>(this ObjectTranslator translator, RealStatePtr L, int index, out <%=full_type_name%> val)
        {
		    LuaTypes type = LuaAPI.lua_type(L, index);
            if (type == LuaTypes.LUA_TUSERDATA )
            {
			    val = (<%=full_type_name%>)XLua.Binding.BindingApi.FastGetCSObj(translator, L, index);
            }
			else if (type == LuaTypes.LUA_TTABLE)
			{
			    val = new <%=full_type_name%>();
				int top = LuaAPI.lua_gettop(L);<%if type_info.Type.IsValueType and type_info.Fields.Length > 0 then%>
				int __xlua_set = 0;<%end%>
				<%ForEachCsList(type_info.Fields, function(fieldInfo)%>
				if (Utils.LoadField(L, index, "<%=fieldInfo.Name%>"))
				{<%if type_info.Type.IsValueType then%>
					__xlua_set++;<%end%>
					translator.<%=StructGetName(fieldInfo.FieldType)%>(L, top + 1, out val.<%=fieldInfo.Name%>);
				}
				LuaAPI.lua_pop(L, 1);
				<%end)%><%if type_info.Type.IsValueType and type_info.Fields.Length > 0 then%>
				// XLUA_STRICT_STRUCT_TABLES: a table that sets none of the fields raises.
				if (__xlua_set == 0) Utils.StructTableSetNoField(typeof(<%=full_type_name%>));<%end%>
			}<%if not type_info.Type.IsValueType then%>
            else if (type == LuaTypes.LUA_TNIL || type == LuaTypes.LUA_TNONE)
            {
                val = null;
            }<%end%>
            else
            {
                throw new Exception("can not cast " + LuaAPI.lua_type(L, index) + " to " + typeof(<%=full_type_name%>));
            }
        }
		<%end)%>

#if UNITY_5_3_OR_NEWER
        [UnityEngine.RuntimeInitializeOnLoadMethod(UnityEngine.RuntimeInitializeLoadType.SubsystemRegistration)]
#endif
#if UNITY_EDITOR
        [UnityEditor.InitializeOnLoadMethod]
#endif
        public static void Register()
        {
            var bindings = new XLua.Binding.AssemblyBindings(typeof(WrapPusher).Assembly.GetName().Name + "/XLua.CSObjectWrap.WrapPusher", XLua.Binding.BindingApi.Version);
			<%ForEachCsList(purevaluetypes, function(type_info)
            if not type_info.Type.IsValueType then return end
            local full_type_name = CsFullTypeName(type_info.Type)
            local v_name = CSVariableName(type_info.Type)%>
            bindings.Struct<<%=full_type_name%>>(Push<%=v_name%>, Get<%=v_name%>, Update<%=v_name%>);
            bindings.ArrayAccessor<<%=full_type_name%>>((translator, L, array, index) => translator.Push<%=v_name%>(L, array[index]),
                (translator, L, array, array_idx, obj_idx) => translator.Get<%=v_name%>(L, obj_idx, out array[array_idx]));<%
			end)%>
			<%ForEachCsList(tableoptimzetypes, function(type_info)
            local full_type_name = CsFullTypeName(type_info.Type)
            local v_name = CSVariableName(type_info.Type)%>
            bindings.Caster<<%=full_type_name%>>(Get<%=v_name%>);
            bindings.ArrayAccessor<<%=full_type_name%>>(null, (translator, L, array, array_idx, obj_idx) => translator.Get<%=v_name%>(L, obj_idx, out array[array_idx]));<%
			end)%>
            XLua.Binding.BindingApi.Register(bindings);
        }
    }
}
