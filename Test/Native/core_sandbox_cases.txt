# The sandbox rules of LuaScript (docs/design/core-luascript.md, sections 3.9
# and 4) as cases for xlua_core_call (XLUA_CALL_CHUNK). Test/Native/core_smoke.c and the
# core_sandbox check of Test/Native/native_regression.py both run this file
# against a state from xlua_core_open (the safe libraries, no limits), in order.
#
# A case starts with "--- name" and may set, on the lines after it:
#   --- env: NAME       run in the environment NAME (created on first use;
#                       "globals" is the default environment, handle 0);
#                       without it every case gets a new environment
#   --- status: N       the expected status (0 XLUA_OK, 2 XLUA_ERRRUN,
#                       3 XLUA_ERRSYNTAX, 4 XLUA_ERRMEM, 10 XLUA_LIMIT_INSTR);
#                       default 0
#   --- expect: TEXT    the first result as text ("nil" without one), or the
#                       message and traceback of an error, contains TEXT
#   --- instructions: N an instruction limit for this case (small slices)
#   --- require: on     enable require from this case on
#   --- libs: buffer    run in a second state that also has the buffer library
#                       (LuaLibs.Buffer), with its own environments; skipped
#                       when the library is built without it (CMake XLUA_BUFFER)
# The rest, up to the next "---" or "===" line, is the chunk. The cases see a
# resolver (the harness) with these modules: "missing" is not found;
# "broken" fails with "resolver failed"; "flaky" is a syntax error on the
# first request and returns "fixed" afterwards; "named" has the chunk name
# "custom/named.lua"; every "=== module NAME" section below is a module.
# A chunk that is "@BINARY@" stands for the first bytes of a Lua binary chunk.
# Lines starting with "#" before the first case are comments.

--- buffer: every read checks its bounds
--- libs: buffer
--- expect: ok

  local read, write = buffer["read" .. name], buffer["write" .. name]
  assert(pcall(read, b, 8 - size), name)
  assert(pcall(write, b, 8 - size, 0), name)
  for _, off in ipairs({8 - size + 1, 8, 9, -1, math.maxinteger, math.mininteger}) do
    local ok, e = pcall(read, b, off)
    assert(not ok and e:find("buffer access out of bounds", 1, true), name .. " read at " .. off)
    ok, e = pcall(write, b, off, 1)
    assert(not ok and e:find("buffer access out of bounds", 1, true), name .. " write at " .. off)
  end
end
return "ok"
